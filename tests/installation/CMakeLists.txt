cmake_minimum_required(VERSION 3.22)
project(calib_install_consumer LANGUAGES CXX)
option(CALIB_CONSUME_AMENT "Resolve the ament package instead of the native package" OFF)
if(CALIB_CONSUME_AMENT)
  find_package(casbot_calib_core REQUIRED CONFIG)
else()
  find_package(casbot_calib REQUIRED CONFIG)
endif()
add_executable(calib_consumer main.cpp)
target_link_libraries(calib_consumer PRIVATE casbot_calib::core)
