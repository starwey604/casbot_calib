cmake_minimum_required(VERSION 3.22)
project(casbot_calib_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(casbot_calib_core REQUIRED)
find_package(casbot_calib_interfaces REQUIRED)
foreach(node mock_host stream_client)
  add_executable(${node} src/${node}.cpp)
  target_include_directories(${node} PRIVATE include)
  ament_target_dependencies(${node} rclcpp casbot_calib_interfaces)
  target_link_libraries(${node} casbot_calib::core)
  target_compile_options(${node} PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
