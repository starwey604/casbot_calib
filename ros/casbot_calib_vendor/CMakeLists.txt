cmake_minimum_required(VERSION 3.22)
project(casbot_calib_vendor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(crb_ros_msg REQUIRED)
find_package(rmw REQUIRED)
find_package(std_srvs REQUIRED)
find_package(casbot_calib_interfaces REQUIRED)
find_package(casbot_calib_core REQUIRED)
add_executable(state_recorder src/state_recorder.cpp)
ament_target_dependencies(state_recorder rclcpp sensor_msgs std_msgs crb_ros_msg rmw)
target_compile_options(state_recorder PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS state_recorder DESTINATION lib/${PROJECT_NAME})
add_executable(position_executor src/position_executor.cpp)
target_include_directories(position_executor PRIVATE include)
ament_target_dependencies(position_executor rclcpp sensor_msgs std_msgs std_srvs crb_ros_msg)
target_compile_options(position_executor PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS position_executor DESTINATION lib/${PROJECT_NAME})
add_executable(mock_sdk_bridge src/mock_sdk_bridge.cpp)
target_include_directories(mock_sdk_bridge PRIVATE include)
ament_target_dependencies(mock_sdk_bridge rclcpp sensor_msgs std_msgs std_srvs crb_ros_msg casbot_calib_interfaces)
target_link_libraries(mock_sdk_bridge casbot_calib::core)
target_compile_options(mock_sdk_bridge PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS mock_sdk_bridge DESTINATION lib/${PROJECT_NAME})
add_executable(sdk_mock_server src/sdk_mock_server.cpp)
target_include_directories(sdk_mock_server PRIVATE include)
ament_target_dependencies(sdk_mock_server rclcpp sensor_msgs std_msgs std_srvs crb_ros_msg casbot_calib_interfaces)
target_link_libraries(sdk_mock_server casbot_calib::core)
target_compile_options(sdk_mock_server PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS sdk_mock_server DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(sdk_position_ramp_test src/sdk_position_ramp_test.cpp)
  target_include_directories(sdk_position_ramp_test PRIVATE include)
  add_test(NAME sdk_position_ramp COMMAND sdk_position_ramp_test)
endif()
ament_package()
