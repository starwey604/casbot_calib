cmake_minimum_required(VERSION 3.22)
project(casbot_calib_vendor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(crb_ros_msg REQUIRED)
find_package(rmw REQUIRED)
find_package(std_srvs REQUIRED)
find_package(casbot_calib_interfaces REQUIRED)
find_package(casbot_calib_core REQUIRED)
find_package(spdlog REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
add_executable(state_recorder src/state_recorder.cpp)
target_include_directories(state_recorder PRIVATE include)
target_link_libraries(state_recorder ZLIB::ZLIB Threads::Threads)
ament_target_dependencies(state_recorder rclcpp sensor_msgs std_msgs crb_ros_msg rmw)
target_compile_options(state_recorder PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS state_recorder DESTINATION lib/${PROJECT_NAME})
add_executable(position_executor src/position_executor.cpp)
target_include_directories(position_executor PRIVATE include)
ament_target_dependencies(position_executor rclcpp sensor_msgs std_msgs std_srvs crb_ros_msg)
target_link_libraries(position_executor casbot_calib::core spdlog::spdlog)
target_compile_options(position_executor PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS position_executor DESTINATION lib/${PROJECT_NAME})
